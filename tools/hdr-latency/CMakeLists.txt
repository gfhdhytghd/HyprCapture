cmake_minimum_required(VERSION 3.19)
project(hdr_latency_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WL REQUIRED IMPORTED_TARGET wayland-client gbm libdrm egl glesv2)
pkg_get_variable(PROTOCOLS wayland-protocols pkgdatadir)
find_program(SCANNER wayland-scanner REQUIRED)
foreach(p color-management linux-dmabuf xdg-shell presentation-time)
  if(p STREQUAL "color-management")
    set(xml "${PROTOCOLS}/staging/color-management/color-management-v1.xml")
  elseif(p STREQUAL "linux-dmabuf")
    set(xml "${PROTOCOLS}/stable/linux-dmabuf/linux-dmabuf-v1.xml")
  else()
    set(xml "${PROTOCOLS}/stable/${p}/${p}.xml")
  endif()
  add_custom_command(OUTPUT ${p}-client.h ${p}-protocol.c
    COMMAND ${SCANNER} client-header ${xml} ${p}-client.h
    COMMAND ${SCANNER} private-code ${xml} ${p}-protocol.c DEPENDS ${xml})
  list(APPEND generated ${p}-client.h ${p}-protocol.c)
endforeach()
add_executable(fp16-probe fp16-probe.cpp ${generated})
target_include_directories(fp16-probe PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(fp16-probe PRIVATE PkgConfig::WL)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
set(CMAKE_AUTOMOC ON)
set(repo "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(ui-benchmark ui-benchmark.cpp
  ${repo}/src/ui/hdr_backdrop.cpp ${repo}/src/ui/hdr_image.cpp
  ${repo}/src/ui/scroll_stitcher.cpp ${repo}/src/ui/annotation_editor.cpp
  ${repo}/src/shared/config.cpp ${repo}/src/shared/protocol.cpp)
target_include_directories(ui-benchmark PRIVATE ${repo}/src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ui-benchmark PRIVATE Qt6::Widgets Qt6::Svg ${OpenCV_LIBS})
add_executable(gpu-benchmark gpu-benchmark.cpp)
target_include_directories(gpu-benchmark PRIVATE ${repo}/src)
target_link_libraries(gpu-benchmark PRIVATE PkgConfig::WL)
